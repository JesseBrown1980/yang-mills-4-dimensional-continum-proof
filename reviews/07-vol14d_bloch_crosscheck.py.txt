#!/usr/bin/env python3
"""Floating-point Bloch-fibre cross-checks behind reviews/07 (24 Nov 2026).

Exploratory numerics, NOT a certificate. Rebuilt from the written definitions in
NOTE-YM4-VOL14C (map and linearization) and NOTE-YM4-VOL14D1 (tree gauge, Z, T);
no repository code is imported. Needs numpy.

  python3 vol14d_bloch_crosscheck.py checks      # construction sanity checks
  python3 vol14d_bloch_crosscheck.py floor 6     # min/max over a 6^4 grid (+ extras)
  python3 vol14d_bloch_crosscheck.py strip 6 0 0.25,0.5,1.0   # sigma_min in a complex strip
  python3 vol14d_bloch_crosscheck.py line        # locate first complex singularities
  python3 vol14d_bloch_crosscheck.py real 8      # real-space ||C_0X|| on an M^4 coarse torus
  python3 vol14d_bloch_crosscheck.py ct          # H offset range and Combes-Thomas rate
  python3 vol14d_bloch_crosscheck.py scalar 3 9  # earlier scalar tent-constraint floor
"""
import itertools
import sys

import numpy as np

B = list(itertools.product((-1, 0, 1), repeat=4))
LINKS = [(x, m) for x in B for m in range(4)]
LIDX = {l: i for i, l in enumerate(LINKS)}
PLAQ = [(x, m, n) for x in B for m in range(4) for n in range(m + 1, 4)]
SITES = {x: i for i, x in enumerate(B)}
NL, NP = len(LINKS), len(PLAQ)
O4 = (0,) * 4


def e(m, s=1):
    return tuple(s if j == m else 0 for j in range(4))


def add(a, b):
    return tuple(p + q for p, q in zip(a, b))


def canon(s):
    X = tuple((a + 1) // 3 for a in s)
    return X, tuple(a - 3 * b for a, b in zip(s, X))


def link_term(s, m):
    X, loc = canon(s)
    return X, LIDX[(loc, m)]


def path(z):  # coordinate-ordered shortest path from the centre
    out = []
    for i, v in enumerate(z):
        out += [(i, 1 if v > 0 else -1)] * abs(v)
    return out


def walk(acc, start, steps, w):
    x = start
    for m, sg in steps:
        if sg > 0:
            key, sign, x = (x, m), 1, add(x, e(m))
        else:
            x = add(x, e(m, -1))
            key, sign = (x, m), -1
        acc[key] = acc.get(key, 0.0) + sign * w
    return acc


def L_row(mu):  # linearization of the VOL14-C map, b = 3
    acc, y = {}, e(mu, 3)
    for z in B:
        ps = set(itertools.permutations(path(z)))
        walk(acc, z, [(mu, 1)] * 3, 1 / 81)
        for p in ps:
            walk(acc, O4, p, 1 / 81 / len(ps))
            walk(acc, y, p, -1 / 81 / len(ps))
    return {k: v for k, v in acc.items() if abs(v) > 1e-15}


def offsets(entries, nrows, ncols=NL):
    mats = {}
    for r, X, c, v in entries:
        mats.setdefault(X, np.zeros((nrows, ncols)))[r, c] += v
    return mats


LROWS = [L_row(m) for m in range(4)]
Loff = offsets([(m, *link_term(s, i), v) for m, r in enumerate(LROWS) for (s, i), v in r.items()], 4)
Dent = []
for p, (x, m, n) in enumerate(PLAQ):
    for s, d, sg in ((x, m, 1), (add(x, e(m)), n, 1), (add(x, e(n)), m, -1), (x, n, -1)):
        Dent.append((p, *link_term(s, d), sg))
Doff = offsets(Dent, NP)
Goff = {}
for li, (x, m) in enumerate(LINKS):
    for s, sg in ((add(x, e(m)), 1), (x, -1)):
        X, loc = canon(s)
        Goff.setdefault(X, np.zeros((NL, 81)))[li, SITES[loc]] += sg

tree = set()
for z in B:
    tree |= {LIDX[k] for k in walk({}, O4, path(z), 1.0)}
cross = [[LIDX[(x, m)] for x in B if x[m] == 1] for m in range(4)]
R = np.zeros((NL, 4))
for m in range(4):
    R[cross[m], m] = 1.0
Cz = np.zeros((84, NL))
for j, t in enumerate(sorted(tree)):
    Cz[j, t] = 1
for m in range(4):
    Cz[80 + m, cross[m]] = 1
Zb = np.linalg.svd(Cz)[2][84:].T  # orthonormal basis of Z (324 x 240)


def sym(off, k):
    return sum(M * np.exp(1j * np.dot(k, X)) for X, M in off.items())


def T_of(k):
    return (np.eye(NL) - R @ sym(Loff, k)) @ Zb


def fibre(k):
    TZ = T_of(k)
    DT = sym(Doff, k) @ TZ
    H, M = DT.conj().T @ DT, TZ.conj().T @ TZ
    Li = np.linalg.inv(np.linalg.cholesky(M))
    return (np.linalg.eigvalsh(H)[[0, -1]], np.linalg.eigvalsh(Li @ H @ Li.conj().T)[0],
            np.linalg.eigvalsh(M)[-1])


def Hc(k):  # analytic continuation of H(k) to complex k
    Tp, Tm = T_of(k), T_of(-k)
    return (sym(Doff, -k) @ Tm).T @ (sym(Doff, k) @ Tp)


def smin(kr, d, eta):
    k = np.array(kr, complex)
    k[d] += 1j * eta
    return np.linalg.svd(Hc(k), compute_uv=False)[-1]


def checks():
    assert len(tree) == 80 and Zb.shape == (324, 240)
    for r in LROWS:
        v = np.array(list(r.values()))
        assert len(r) == 459 and abs((v ** 2).sum() - 277 / 729) < 1e-12
        assert abs(np.abs(v).sum() - 205 / 27) < 1e-12
    rng = np.random.default_rng(0)
    for _ in range(5):
        k = rng.uniform(-np.pi, np.pi, 4)
        Lk, Gk = sym(Loff, k), sym(Goff, k)
        phi = rng.normal(size=81) + 1j * rng.normal(size=81)
        c = SITES[O4]
        assert np.allclose(Lk @ R, np.eye(4))
        assert np.allclose(Lk @ Gk @ phi, [(np.exp(1j * k[m]) - 1) * phi[c] for m in range(4)])
        assert np.allclose(sym(Doff, k) @ Gk, 0)
    print("checks PASS: 459/277-729/205-27 rows, 80 tree links, dim Z=240, LR=I, L d phi=d_coarse phi, dd=0")


def floor(n):
    rng = np.random.default_rng(0)
    g = np.linspace(-np.pi, np.pi, n, endpoint=False)
    grid = list(itertools.product(g, repeat=4)) + [tuple(v) for v in rng.uniform(-np.pi, np.pi, (500, 4))]
    best = {"lam_min(H)": (9, None), "lam_max(H)": (0, None),
            "min ||dA||^2/||A||^2": (9, None), "max ||T||^2": (0, None)}
    for k in grid:
        (h0, h1), g0, t = fibre(np.array(k, float))
        for key, v, better in (("lam_min(H)", h0, h0 < best["lam_min(H)"][0]),
                               ("lam_max(H)", h1, h1 > best["lam_max(H)"][0]),
                               ("min ||dA||^2/||A||^2", g0, g0 < best["min ||dA||^2/||A||^2"][0]),
                               ("max ||T||^2", t, t > best["max ||T||^2"][0])):
            if better:
                best[key] = (v, k)
    for key, (v, k) in best.items():
        print(f"{key}: {v:.6g} at k={np.round(k, 3)}")


def strip(n, d, etas):
    g = np.linspace(-np.pi, np.pi, n, endpoint=False)
    for eta in etas:
        v, k = min((smin(kr, d, eta), kr) for kr in itertools.product(g, repeat=4))
        print(f"e{d + 1} eta={eta}: min sigma_min={v:.4g} at Re k={np.round(k, 3)}")


def line():
    for kr, d in (((np.pi, 0, 0, 0), 0), ((np.pi,) * 4, 0), ((0, 0, 0, np.pi), 3)):
        etas = np.linspace(1.0, 2.0, 21)
        i = int(np.argmin([smin(kr, d, t) for t in etas]))
        fine = np.linspace(etas[max(i - 1, 0)], etas[min(i + 1, 20)], 41)
        fv = [smin(kr, d, t) for t in fine]
        j = int(np.argmin(fv))
        print(f"Re k={np.round(kr, 3)} along e{d + 1}: sigma_min={fv[j]:.3g} at eta={fine[j]:.4f}")


def real(M):
    ks = [np.array(v) * 2 * np.pi / M for v in itertools.product(range(M), repeat=4)]
    tg = [(r, 0, 0, 0) for r in range(M // 2 + 1)] + [(0, 0, 0, r) for r in range(1, M // 2 + 1)]
    tg += [(1, 1, 0, 0), (1, 1, 1, 1), (2, 2, 0, 0), (2, 2, 2, 2)]
    acc = {X: 0 for X in tg}
    for k in ks:
        T = T_of(k)
        DT = sym(Doff, k) @ T
        Ck = T @ np.linalg.solve(DT.conj().T @ DT, T.conj().T)
        for X in tg:
            acc[X] = acc[X] + Ck * np.exp(1j * np.dot(k, X))
    for X in tg:
        print(X, f"||C_0X|| = {np.linalg.norm(acc[X] / len(ks), 2):.4g}")


def ct(lam=0.032644):
    Toff = {X: -R @ M for X, M in Loff.items()}
    Toff[O4] = Toff.get(O4, 0) + np.eye(NL)
    Toff = {X: M @ Zb for X, M in Toff.items()}
    DT = {}
    for a, Da in Doff.items():
        for b, Tb in Toff.items():
            DT[add(a, b)] = DT.get(add(a, b), 0) + Da @ Tb
    HX = {}
    for Y, A in DT.items():
        for Y2, B2 in DT.items():
            X = tuple(q - p for p, q in zip(Y, Y2))
            HX[X] = HX.get(X, 0) + A.T @ B2
    HX = {X: M for X, M in HX.items() if np.abs(M).max() > 1e-13}
    print("H offset blocks:", len(HX), " max l1 range:", max(sum(map(abs, X)) for X in HX))
    nrm = {X: np.linalg.norm(M, 2) for X, M in HX.items()}
    lo, hi = 0.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        lo, hi = (mid, hi) if sum(v * (np.exp(mid * abs(X[0])) - 1) for X, v in nrm.items()) < lam else (lo, mid)
    print(f"Combes-Thomas along e1 with lam_min={lam}: eta < {lo:.5f} per coarse step")


def scalar(b, m):  # earlier scalar check; the constraint a.phi=0 is bilinear, so complement conj(a)
    d, off = 4, b // 2
    Rr = list(itertools.product(range(b), repeat=d))
    idx = {r: i for i, r in enumerate(Rr)}
    W = [((z0 - off,) + tuple(t - off for t in zt), min(z0 + 1, 2 * b - 1 - z0, b))
         for z0 in range(2 * b - 1) for zt in itertools.product(range(b), repeat=3)]
    best = (9, None)
    for k in itertools.product(np.linspace(-np.pi, np.pi, m), repeat=4):
        k = np.array(k)
        H = np.zeros((len(Rr),) * 2, complex)
        for r in Rr:
            H[idx[r], idx[r]] += 2 * d
            for nu in range(d):
                for s in (1, -1):
                    rr, ph = list(r), 1
                    rr[nu] += s
                    if rr[nu] == b:
                        rr[nu], ph = 0, np.exp(1j * k[nu])
                    if rr[nu] == -1:
                        rr[nu], ph = b - 1, np.exp(-1j * k[nu])
                    H[idx[r], idx[tuple(rr)]] -= ph
        a = np.zeros(len(Rr), complex)
        for z, w in W:
            a[idx[tuple(np.mod(z, b))]] += w * np.exp(1j * np.dot(k, np.floor_divide(z, b)))
        Q = np.linalg.qr(np.column_stack([(a / np.linalg.norm(a)).conj()] + list(np.eye(len(Rr)))))[0][:, 1:len(Rr)]
        v = np.linalg.eigvalsh(Q.conj().T @ H @ Q)[0]
        if v < best[0]:
            best = (v, k)
    print(f"b={b}: lambda_min={best[0]:.5f} at k={np.round(best[1], 3)}; 4sin^2(pi/2b)={4 * np.sin(np.pi / (2 * b)) ** 2:.5f}")


if __name__ == "__main__":
    cmd, a = sys.argv[1], sys.argv[2:]
    checks()
    if cmd == "floor":
        floor(int(a[0]) if a else 6)
    elif cmd == "strip":
        strip(int(a[0]), int(a[1]), [float(x) for x in a[2].split(",")])
    elif cmd == "line":
        line()
    elif cmd == "real":
        real(int(a[0]) if a else 8)
    elif cmd == "ct":
        ct()
    elif cmd == "scalar":
        scalar(int(a[0]), int(a[1]))
